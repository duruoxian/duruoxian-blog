综合题解析

综合1：思想——层序遍历，空指针也入队；一旦出队遇到NULL，其后所有出队结点必须全为NULL，否则非完全二叉树。

int IsComplete(BiTree T){
    Queue Q; BiTree p;
    if(!T) return 1;
    InitQueue(Q); EnQueue(Q,T);
    while(!QueueEmpty(Q)){
        DeQueue(Q,p);
        if(p){ EnQueue(Q,p->lchild); EnQueue(Q,p->rchild); }
        else
            while(!QueueEmpty(Q)){
                DeQueue(Q,p);
                if(p) return 0;   /* 空后还有非空 → 非完全 */
            }
    }
    return 1;
}   /* O(n), O(n) */
